```python
            toggle_source,
            inputs=[source_type],
            outputs=[pdf_group, pubmed_group]
        )
        
        # Submit handler
        submit_btn.click(
            process_source,
            inputs=[source_type, pdf_file, search_keywords, max_results],
            outputs=[status, output_mcqs]
        )
    
    return interface
```

### 2. LangChain Orchestration (LangGraph)

```python
from langgraph.graph import StateGraph, END
from typing import TypedDict, List, Dict, Any

class MCQState(TypedDict):
    source_type: str  # "PDF" | "PUBMED"
    source_id: str     # UUID or PMID
    full_text: str
    extracted_relations: List[Dict]
    validated_relations: List[Dict]
    stored_triples: List[int]  # triple_ids
    generated_mcqs: List[Dict]
    verified_mcqs: List[Dict]

def create_workflow():
    """Create LangGraph workflow."""
    
    workflow = StateGraph(MCQState)
    
    # Add nodes
    workflow.add_node("extract_text", extract_text_node)
    workflow.add_node("extract_relations", extract_relations_node)
    workflow.add_node("validate_schema", validate_schema_node)
    workflow.add_node("store_triples", store_triples_node)
    workflow.add_node("generate_mcq", generate_mcq_node)
    workflow.add_node("verify_triple", verify_triple_node)
    
    # Define flow
    workflow.set_entry_point("extract_text")
    workflow.add_edge("extract_text", "extract_relations")
    workflow.add_edge("extract_relations", "validate_schema")
    workflow.add_edge("validate_schema", "store_triples")
    workflow.add_edge("store_triples", "generate_mcq")
    workflow.add_edge("generate_mcq", "verify_triple")
    workflow.add_edge("verify_triple", END)
    
    return workflow.compile()
```

```python
def build_relation_extraction_prompt(full_text: str, schema: Dict) -> str:
    """Build prompt for relation extraction with evidence snippets."""
    
    relations_list = "\n".join([
        f"- {rel['id']}: {rel['description']}"
        for rel in schema['relations']
    ])
    
    return f"""Extract medical relationships from this text and return exact evidence.

**Full Text:**
"{full_text}"

**Allowed Relation Types (Schema):**
{relations_list}

**Task:**
For each relationship found:
1. Extract triple: (head_entity, relation_type, tail_entity)
2. Copy EXACT 2-4 sentence snippet where relationship appears
3. Provide location (paragraph, sentence numbers, character positions)

**CRITICAL - Evidence Snippet Rules:**
- Include 2-4 COMPLETE sentences containing the relationship
- Start from sentence introducing the relationship
- Include enough context to make relationship clear
```

### 6. Triple Verification (3 Checks)

```python
def verify_triple(triple_id: int, mcq_data: Dict, db_conn) -> Dict:
    """Verify triple exists and is valid."""
    
    # Check 1: Triple exists in DB
    triple = db_conn.execute("""
        SELECT head_entity, relation, tail_entity, source_id, evidence_snippet
        FROM triples WHERE triple_id = ?
    """, (triple_id,)).fetchone()
    
    if not triple:
        return {
            "verified": False,
            "reason": "Triple not found in database"
        }
    
    head, relation, tail, source_id, evidence = triple
    
    # Check 2: Triple matches MCQ (enforcement)
    if mcq_data["triple_used"]["head"] != head or \
       mcq_data["triple_used"]["relation"] != relation or \
       mcq_data["triple_used"]["tail"] != tail:
        return {
            "verified": False,
            "reason": "LLM changed the triple - enforcement failed"
        }
    
    # Check 3: Evidence exists in source (LangChain)
    source_text = db_conn.execute("""
        SELECT full_text FROM sources WHERE source_id = ?
    """, (source_id,)).fetchone()[0]
    
    verification_result = langchain_verify_evidence(
        claim=f"{head} {relation} {tail}",
        evidence_snippet=evidence,
        source_text=source_text
    )
    
    return {
        "verified": verification_result["verified"],
        "triple_exists": True,
        "triple_matches": True,
        "evidence_in_source": verification_result["verified"],
        "confidence": verification_result["confidence"]
    }
```

### 7. LangChain Evidence Verification

```python
from langchain.chains import LLMChain
from langchain.prompts import PromptTemplate
from langchain.llms import OpenAI

def langchain_verify_evidence(claim: str, evidence_snippet: str, source_text: str) -> Dict:
    """Use LangChain to verify evidence exists in source."""
    
    prompt = PromptTemplate(
        input_variables=["claim", "evidence_snippet", "source_text"],
        template="""Verify if this evidence snippet exists in the source text.

**Claim:** {claim}

**Claimed Evidence Snippet:**
"{evidence_snippet}"

**Full Source Text:**
"{source_text}"

**Verification Task:**
1. Check if the evidence snippet appears in source text (exact or near-exact match)
2. Verify the claim is accurately stated in the snippet
3. Confirm no contradictions or negations

**Output JSON:**
{{
    "verified": true/false,
    "confidence": 0.0-1.0,
    "snippet_found": true/false,
    "reasoning": "Brief explanation"
}}
"""
    )
    
    chain = LLMChain(llm=OpenAI(temperature=0), prompt=prompt)
    result = chain.run(
        claim=claim,
        evidence_snippet=evidence_snippet,
        source_text=source_text
    )
    
    return json.loads(result)
```

### 9. Iterative Refinement (Same Prompt + Feedback)

```python
def refine_mcq_with_feedback(triple_id: int, feedback: str, db_conn) -> Dict:
    """Refine MCQ using same prompt structure + user feedback."""
    
    # Get original triple
    triple = db_conn.execute("""
        SELECT head_entity, relation, tail_entity, evidence_snippet
        FROM triples WHERE triple_id = ?
    """, (triple_id,)).fetchone()
    
    head, relation, tail, evidence = triple
    
    # Build refined prompt (same structure + feedback)
    prompt = f"""Generate a BETTER medical MCQ from this relationship, incorporating user feedback.

**Relationship (DO NOT CHANGE):**
- Head: {head}
- Relation: {relation}
- Tail: {tail}

**Evidence Snippet:**
"{evidence}"

**User Feedback:**
"{feedback}"

**Requirements:**
1. MCQ MUST test: {head} {relation} {tail} (unchanged)
2. Incorporate user feedback to improve the question
3. Maintain medical accuracy and educational value
4. Correct answer must be supported by evidence

**Output JSON (same format as before):**
{{
    "stem": "Improved question...",
    "options": [...],
    "explanation": "...",
    "triple_used": {{"head": "{head}", "relation": "{relation}", "tail": "{tail}"}}
}}
"""
    
    # Call LLM
    response = llm.generate(prompt)
    mcq_data = json.loads(response)
    
    # Verify triple still matches (enforcement)
    if mcq_data["triple_used"]["head"] != head or \
       mcq_data["triple_used"]["relation"] != relation or \
       mcq_data["triple_used"]["tail"] != tail:
        raise ValueError("Triple changed during refinement - rejected")
    
    return mcq_data
```

## Database Schema (Simplified)

```sql
-- Sources
CREATE TABLE sources (
    source_id TEXT PRIMARY KEY,      -- "PMID:12345678" or "UUID:abc12345"
    source_type TEXT,
    full_text TEXT NOT NULL,
    metadata_json TEXT,
    created_at TIMESTAMP
);

-- Triples (with evidence snippets)
CREATE TABLE triples (
    triple_id INTEGER PRIMARY KEY,
    source_id TEXT REFERENCES sources(source_id),
    head_entity TEXT NOT NULL,
    relation TEXT NOT NULL,
    tail_entity TEXT NOT NULL,
    evidence_snippet TEXT NOT NULL,  -- 2-4 sentences
    location_paragraph INTEGER,
    location_sentence_start INTEGER,
    location_sentence_end INTEGER,
    location_char_start INTEGER,
    location_char_end INTEGER,
    confidence REAL,
    created_at TIMESTAMP
);

-- MCQs
CREATE TABLE mcqs (
    mcq_id INTEGER PRIMARY KEY,
    triple_id INTEGER REFERENCES triples(triple_id),
    stem TEXT NOT NULL,
    options_json TEXT NOT NULL,
    explanation TEXT,
    citation TEXT,
    triple_used_json TEXT,           -- Stored for verification
    verification_status TEXT,
    verification_confidence REAL,
    status TEXT DEFAULT 'pending',    -- pending/approved/rejected/refining
    user_feedback TEXT,
    created_at TIMESTAMP
);
```

## Summary

**Architecture Features:**
1. ✅ Gradio UI for input (PDF/PubMed via Tavily)
2. ✅ LangGraph orchestration
3. ✅ LLM extracts relations with 2-4 sentence evidence snippets
4. ✅ Schema validation (local check)
5. ✅ Triple stored in DB with evidence
6. ✅ MCQ generation with triple enforcement
7. ✅ Triple verification (3 checks: exists, matches, evidence in source)
8. ✅ Human review UI with provenance display
9. ✅ Iterative refinement (same prompt + feedback)

**Key Innovations:**
- **Evidence Snippets**: LLM returns exact 2-4 sentences
- **Triple Enforcement**: MCQ generation cannot change triple
- **LangChain Verification**: Checks evidence exists in source
- **Iterative Refinement**: Same prompt structure, adds feedback
- **Full Provenance**: Every MCQ shows triple, evidence, citation

This architecture is LLM-optimized and provenance-first. Ready for implementation!
